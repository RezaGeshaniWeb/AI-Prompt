// By declaring the function as "async", it automatically returns a Promise.
// This allows us to remove the "new Promise" wrapper and directly throw errors or return values.
async function checkPositiveNumber(number) {
  if (typeof number !== "number") {
    // Throwing an error in an async function is equivalent to rejecting a Promise.
    throw new Error("ورودی باید یک عدد باشد.");
  }

  if (number >= 0) {
    // Returning a value in an async function is equivalent to resolving a Promise.
    return "عدد مثبت یا صفر است.";
  } else {
    // Rejecting the Promise with an error for negative values.
    throw new Error("عدد منفی است.");
  }
}
