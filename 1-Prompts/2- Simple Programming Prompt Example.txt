/**
 * Checks if a given number is Even or Odd.
 * Handles invalid inputs by returning null.
 * 
 * @param {any} input - The value to check.
 * @returns {string|null} "Even", "Odd", or null for invalid inputs.
 */
function checkEvenOrOdd(input) {
    // بررسی معتبر بودن ورودی: ورودی باید عدد باشد و مقدار NaN یا بی‌نهایت (Infinity) نباشد.
    // از آنجا که ورودی باید عدد صحیح (Integer) باشد، بررسی صحیح بودن (isInteger) انجام می‌شود.
    if (!Number.isInteger(input)) {
        return null; // بازگرداندن null برای ورودی‌های نامعتبر
    }

    // استفاده از عملگر باقیمانده (Modulo) برای تشخیص زوج یا فرد بودن
    return input % 2 === 0 ? "Even" : "Odd";
}
